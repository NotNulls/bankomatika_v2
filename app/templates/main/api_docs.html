Python

import os
import requests

response = requests.post(
    "{{ api_endpoint }}",
    headers={"Authorization": "Bearer " + os.environ["BSK_API_KEY"]},
    files={"file": open("statement.pdf", "rb")},
    timeout=120,
)
response.raise_for_status()
data = response.json()
for row in data["rows"]:
    print(row["date"], row["description"], row["debit"], row["credit"])
